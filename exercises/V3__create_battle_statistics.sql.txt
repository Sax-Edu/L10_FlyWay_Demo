-- Skapar en separat statistikrad per Pokémon.

CREATE TABLE battle_statistics (
    statistics_id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    pokemon_id BIGINT NOT NULL UNIQUE REFERENCES pokemon(pokemon_id),
    wins INTEGER NOT NULL DEFAULT 0 CHECK (wins >= 0),
    losses INTEGER NOT NULL DEFAULT 0 CHECK (losses >= 0)
);
